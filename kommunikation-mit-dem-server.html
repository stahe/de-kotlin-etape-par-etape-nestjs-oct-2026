Der Befehl [adb reverse tcp:8080 tcp:8080] verbindet Port 8080 des Telefons über das USB-Kabel (oder direkt beim Emulator) mit Port 8080 des Rechners. Das ist die Standardlösung: [http://localhost:8080]. Der Befehl muss bei jedem Start des Emulators erneut ausgeführt werden;

// ex22/Ex22Activity.kt - eine erste HTTP-Anfrage an den Server
// -------------------------------------------------------------------------
// Der JSON-Server von RdvMedecins muss laufen (Port 8080). Wir fordern von ihm
// ein CAPTCHA an: GET /api/captcha -> { "id": "...", "svg": "<svg ...>" }.
//
//   - OkHttp, der am häufigsten genutzte HTTP-Client unter Android: Wir beschreiben die
//     Anfrage (Request), der Client führt sie aus (newCall(...).execute()) und
//     liefert die Antwort: ihren Statuscode (code), ihre Header
//     (headers), ihren Body (body);
//   - execute() BLOCKIERT bis zur Antwort: Android verbietet den Aufruf
//     im UI-Thread. withContext(Dispatchers.IO)
//     führt ihn in einem anderen Thread aus, und die Coroutine WARTET auf das Ergebnis;
//   - Json.parseToJsonElement(text): der JSON-Body -> Kotlin-Objekte;
//   - die Adresse des Servers hängt davon ab, wo die Anwendung läuft
//     (Emulator, Smartphone ...): vgl. common/Config.kt;
//   - drei anzuzeigende Zustände: wird geladen, erfolgreich, fehlerhaft;
//   - das CAPTCHA-Bild ist SVG: vgl. common/SvgImage.kt.
// =========================================================================

// was GET /api/captcha zurückgibt
data class Captcha(val id: String, val svg: String)

// ein HTTP-Client: ein aufwendig zu erzeugendes Objekt, das nur EINMAL erzeugt wird
val client = OkHttpClient()

// die Anfrage: eine SUSPEND-Funktion, die das CAPTCHA liefert oder eine Exception auslöst
suspend fun readCaptcha(): Captcha = withContext(Dispatchers.IO) {
    val request = Request.Builder().url("${Config.serverUrl}/api/captcha").build()
    client.newCall(request).execute().use { response ->
        // OkHttp löst bei einer Fehlerantwort (4xx,
        // 5xx) KEINE Exception aus: der Statuscode wird geprüft
        if (!response.isSuccessful) {
            throw Exception("the server responded ${response.code} ${response.message}")
        }
        val json = Json.parseToJsonElement(response.body!!.string()).jsonObject
        Captcha(json["id"]!!.jsonPrimitive.content, json["svg"]!!.jsonPrimitive.content)
    }
}

@Composable
fun CaptchaPage() {
    var captcha by remember { mutableStateOf<Captcha?>(null) }
    var loading by remember { mutableStateOf(true) }
    var error by remember { mutableStateOf<String?>(null) }
    var requestCount by remember { mutableIntStateOf(0) }
    // die Nummer der ANFORDERUNG: Eine Änderung startet den Effekt unten neu
    var demand by remember { mutableIntStateOf(0) }

    // die Anfrage, sobald der Bildschirm erscheint (demand = 0), danach bei jeder
    // neuen Anforderung. Trifft eine Anforderung ein, bevor die Antwort auf die
    // vorherige da ist, wird die vorherige Coroutine ABGEBROCHEN: ihre veraltete Antwort
    // wird nie angezeigt
    LaunchedEffect(demand) {
        loading = true
        error = null
        try {
            captcha = readCaptcha()
            requestCount++
        } catch (e: CancellationException) {
            throw e // die Coroutine wurde abgebrochen (neue Anforderung): nichts anzuzeigen
        } catch (e: Exception) {
            // überhaupt keine Antwort (Server gestoppt, falsche Adresse ...) oder Fehler
            error = e.message ?: e.toString()
            captcha = null
        }
        loading = false
    }

    ExamplePage("Example 22 - a first HTTP request") {
        Text("Server: ${Config.serverUrl}", color = Color.Gray)
        Spacer(Modifier.height(12.dp))
        if (loading) LinearProgressIndicator(Modifier.fillMaxWidth())
        error?.let {
            Card(colors = CardDefaults.cardColors(containerColor = Color(0xFFFFEBEE))) {
                Text("Unable to get a captcha: $it.\nIs the RdvMedecins JSON server running (port 8080)?", Modifier.padding(12.dp))
            }
        }
        val c = captcha
        if (c != null && !loading) {
            SvgImage(
                c.svg,
                height = 60.dp,
                modifier = Modifier.border(1.dp, Color(0xFFE0E0E0), RoundedCornerShape(6.dp)).padding(4.dp),
            )
            Spacer(Modifier.height(8.dp))
            Text("id: ${c.id}", fontFamily = FontFamily.Monospace, fontSize = 11.sp)
        }
        Spacer(Modifier.height(12.dp))
        Button(onClick = { demand++ }, enabled = !loading) { Text("Another image") }
        Spacer(Modifier.height(8.dp))
        Text("$requestCount successful request(s)", color = Color.Gray)
    }
}
    • Zeile 66: ein HTTP-Client, ein Objekt, dessen Erzeugung aufwendig ist und das nur EINMAL erzeugt wird;
    • Zeilen 69–71: die Anfrage, eine SUSPEND-Funktion. Die Anfrage ([Request]) wird beschrieben; der Client führt sie aus ([newCall(...).execute()]) und liefert die Antwort: ihren Statuscode, ihre Header und ihren Body. [execute()] BLOCKIERT bis zur Antwort: [Android] verbietet den Aufruf im Ausführungs-Thread des Bildschirms ([NetworkOnMainThreadException]). [withContext(Dispatchers.IO)] führt sie in einem für I/O reservierten Thread aus, und die Coroutine WARTET auf das Ergebnis;
    • Zeilen 74–75: [OkHttp] wirft bei einer Fehlerantwort (4xx, 5xx) keine Exception; stattdessen prüft man den Statuscode. Es wirft jedoch eine [IOException], wenn überhaupt keine Antwort eintrifft (Server nicht erreichbar, falsche Adresse);

    • Zeilen 103–116: Die Anfrage wird von einem [LaunchedEffect] ausgelöst, dessen Schlüssel eine ANFRAGENUMMER ist: Die Schaltfläche „Another image“ erhöht sie, wodurch der Effect neu gestartet wird. Trifft eine Anfrage ein, bevor die vorherige geantwortet hat, wird die vorherige Coroutine ABGEBROCHEN: Ihre inzwischen veraltete Antwort wird nie angezeigt;

import android.graphics.Bitmap
import android.graphics.Canvas
import androidx.compose.foundation.Image
import androidx.compose.foundation.layout.height
import androidx.compose.runtime.Composable
import androidx.compose.runtime.remember
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.asImageBitmap
import androidx.compose.ui.layout.ContentScale
import androidx.compose.ui.unit.Dp
import androidx.compose.ui.unit.dp
import com.caverock.androidsvg.SVG

@Composable
fun SvgImage(svg: String, modifier: Modifier = Modifier, height: Dp = 50.dp, description: String? = null) {
    // die Zeichnung wird nur neu erstellt, wenn sich der SVG-Text ändert (remember(svg))
    val image = remember(svg) {
        val document = SVG.getFromString(svg)
        // 3-mal größer als die "natürliche" Größe gezeichnet: ein scharfes Bild
        val scale = 3f
        val width = (document.documentWidth.takeIf { it > 0 } ?: 150f)
        val tall = (document.documentHeight.takeIf { it > 0 } ?: 50f)
        val bitmap = Bitmap.createBitmap((width * scale).toInt(), (tall * scale).toInt(), Bitmap.Config.ARGB_8888)
        val canvas = Canvas(bitmap)
        canvas.scale(scale, scale)
        document.renderToCanvas(canvas)
        bitmap.asImageBitmap()
    }
    Image(image, contentDescription = description, contentScale = ContentScale.Fit, modifier = modifier.height(height))
}

    // =========================================================================
// common/HttpClient.kt - ein HTTP-Client, der COOKIES verwaltet
// -------------------------------------------------------------------------
// Der RdvMedecins-Server legt das JWT-Token in einem COOKIE (httpOnly) ab.
// In einem Browser speichert der Browser die Cookies und sendet sie
// bei jeder Anfrage zurück: Der Code der Anwendung muss sich um nichts kümmern.
// Auf einem Smartphone gibt es KEINEN Browser: Die Anwendung muss
// seine Rolle übernehmen, das heißt:
//   1. den Header "Set-Cookie" jeder Antwort lesen und die Cookies speichern;
//   2. sie im Header "Cookie" jeder Anfrage zurücksenden.
// Die Bibliothek OkHttp erledigt das für uns, wenn wir ihr einen CookieJar geben
// (ein "Cookie Jar"): eine Klasse mit zwei Funktionen, saveFromResponse
// (speichern) und loadForRequest (herausholen). Dieser speichert die Cookies
// auch auf dem Gerät (SharedPreferences): Wir bleiben angemeldet, wenn wir die Anwendung
// beenden und neu starten.
// =========================================================================
package fr.stahe.examples.common

import android.content.Context
import android.content.SharedPreferences
import androidx.core.content.edit
import okhttp3.Cookie
import okhttp3.CookieJar
import okhttp3.HttpUrl
import okhttp3.HttpUrl.Companion.toHttpUrl
import okhttp3.OkHttpClient
import java.util.concurrent.TimeUnit

class DeviceCookieJar(private val storage: SharedPreferences) : CookieJar {
    // Cookie-Name -> das Cookie (sein Wert, seine Domain, sein Ablaufdatum ...)
    private val cookies = mutableMapOf<String, Cookie>()

    init {
        // die bei einer früheren Nutzung gespeicherten Cookies: Jedes
        // wird in Form seines Set-Cookie-Headers gespeichert
        val base = Config.serverUrl.toHttpUrl()
        for (text in storage.getStringSet(KEY, emptySet()).orEmpty()) {
            Cookie.parse(base, text)?.let { cookies[it.name] = it }
        }
    }

    // die Antwort bringt Cookies mit: Wir speichern sie. Ein bereits abgelaufenes Cookie
    // (so löscht der Server es beim Logout) wird entfernt
    @Synchronized
    override fun saveFromResponse(url: HttpUrl, cookies: List<Cookie>) {
        for (c in cookies) {
            if (c.expiresAt < System.currentTimeMillis() || c.value.isEmpty()) {
                this.cookies.remove(c.name)
            } else {
                this.cookies[c.name] = c
            }
        }
        storage.edit { putStringSet(KEY, this@DeviceCookieJar.cookies.values.map { it.toString() }.toSet()) }
    }

    // eine Anfrage geht hinaus: die Cookies, die sie betreffen (Adresse, Pfad) und nicht abgelaufen sind
    @Synchronized
    override fun loadForRequest(url: HttpUrl): List<Cookie> =
        cookies.values.filter { it.expiresAt >= System.currentTimeMillis() && it.matches(url) }

    companion object {
        private const val KEY = "rdvmedecins.cookies"
    }
}

// der HTTP-Client der Beispiele: ein einziger für die ganze Anwendung
fun createHttpClient(context: Context): OkHttpClient = OkHttpClient.Builder()
    .cookieJar(DeviceCookieJar(context.getSharedPreferences("examples", Context.MODE_PRIVATE)))
    // keine Antwort nach 10 Sekunden: Die Anfrage schlägt fehl
    .connectTimeout(10, TimeUnit.SECONDS)
    .readTimeout(10, TimeUnit.SECONDS)
    .build()
    • Zeile 29: Ein [CookieJar] hat zwei Funktionen: [saveFromResponse] (Cookies einer Antwort speichern) und [loadForRequest] (die Cookies für eine Anfrage abrufen);
    • Zeilen 33–38: Beim Start werden Cookies, die während einer früheren Sitzung gespeichert wurden – auf dem Gerät gesichert ([SharedPreferences], eine kleine Schlüssel-Wert-Datei; siehe Beispiel 20) –, in Form ihres Headers [Set-Cookie] wieder abgerufen;
    • Zeilen 44–53: Ein bereits abgelaufenes oder leeres Cookie wird entfernt; so löscht der Server das Cookie bei der Abmeldung; [@Synchronized]: [OkHttp] kann diese Funktionen aus mehreren Threads gleichzeitig aufrufen;
    • Zeilen 67–72: der HTTP-Client aus den Beispielen, mit seinem Jar und seinen Timeouts.

// ex23/Command.kt - eine AKTION, ihr Zustand, ihr Ergebnis
// -------------------------------------------------------------------------
// Eine asynchrone Aktion (Anmelden, Speichern ...) hat einen Zustand,
// den der Bildschirm anzeigen muss: Läuft sie GERADE (Button deaktiviert)? Endete sie
// mit einem FEHLER (Meldung)? Es ist das Muster "Command", auf das
// Wesentliche reduziert:
//
//     val loginCommand = remember { Command() }
//     scope.launch { loginCommand.execute { logIn() } }    // startet die Aktion
//     loginCommand.inProgress, loginCommand.error          // ihr Zustand
//
// Ihre beiden Properties sind von Compose beobachtete Zustände: Der Bildschirm wird
// neu gezeichnet, wenn die Aktion beginnt und wenn sie endet.
// =========================================================================
package fr.stahe.examples.ex23

import androidx.compose.runtime.getValue
import androidx.compose.runtime.mutableStateOf
import androidx.compose.runtime.setValue
import kotlinx.coroutines.CancellationException

class Command {
    var inProgress by mutableStateOf(false)
        private set
    var error by mutableStateOf<String?>(null)
        private set

    // führt die Aktion aus, sofern nicht bereits eine andere läuft
    suspend fun execute(action: suspend () -> Unit) {
        if (inProgress) return // kein doppeltes Absenden (doppeltes Drücken des Buttons)
        inProgress = true
        error = null
        try {
            action()
        } catch (e: CancellationException) {
            throw e // eine abgebrochene Coroutine muss abgebrochen bleiben
        } catch (e: Exception) {
            error = e.message ?: e.toString() // der Fehler wird BEHALTEN, um angezeigt zu werden

// -------------------------------------------------------------------------
// Alle Fehlerantworten des RdvMedecins-Servers haben dieselbe Form:
//   { "statusCode": 409, "key": "ERRORS.LOGIN_TAKEN", "params": {...}, "fields": {...} }

// (Netzwerk ausgefallen, Server gestoppt, falsche Adresse) oder eine Antwort, die

// all das in EINE Exception-Klasse um, ApiError, die den vom Server gesendeten Übersetzungsschlüssel

package fr.stahe.examples.common

import kotlinx.serialization.json.Json
import kotlinx.serialization.json.JsonObject
import kotlinx.serialization.json.JsonPrimitive
import kotlinx.serialization.json.jsonObject

class ApiError(
    val statusCode: Int, // 0: der Server hat nicht geantwortet
    val key: String, // z. B.: "ERRORS.LOGIN_TAKEN"
    val params: Map<String, String> = emptyMap(),
    val fields: Map<String, String> = emptyMap(),
) : Exception("$statusCode $key") {

    override fun toString() = "$statusCode $key"

    // companion object: was zur KLASSE gehört (das "static" von Java)
    companion object {
        // eine Fehlerantwort (4xx, 5xx) -> ein ApiError
        fun fromResponse(statusCode: Int, body: String): ApiError {
            // runCatching: das Ergebnis oder null, wenn der Body kein JSON ist
            // (dann hat nicht unser Server geantwortet)
            val json = runCatching { Json.parseToJsonElement(body).jsonObject }.getOrNull()
            val key = (json?.get("key") as? JsonPrimitive)?.content
            if (json == null || key == null) {
                return ApiError(statusCode, if (statusCode >= 500) "ERRORS.SERVER_UNREACHABLE" else "ERRORS.SERVER_ERROR")
            }
            return ApiError(statusCode, key, strings(json["params"]), strings(json["fields"]))
        }

        // ein JSON-Objekt { "name": "ERRORS..." } -> Map<String, String> (sonst leer)
        private fun strings(json: Any?): Map<String, String> =
            (json as? JsonObject)?.mapValues { (_, v) -> (v as? JsonPrimitive)?.content ?: v.toString() } ?: emptyMap()
    }
}

class Api(private val client: OkHttpClient) {

    suspend fun request(method: String, url: String, body: JsonElement? = null): JsonElement? =
        withContext(Dispatchers.IO) {
            // ein JSON-Body (der Server weist einen POST ohne Body zurück)
            val httpBody = body?.toString()?.toRequestBody("application/json".toMediaType())
            val request = Request.Builder().url("${Config.serverUrl}/api$url").method(method, httpBody).build()
            try {
                // use: die Antwort wird am Ende des Blocks geschlossen (freigegeben)
                client.newCall(request).execute().use { response ->
                    val text = response.body?.string().orEmpty()
                    if (!response.isSuccessful) throw ApiError.fromResponse(response.code, text)
                    if (text.isEmpty()) null else Json.parseToJsonElement(text)
                }
            } catch (e: IOException) {
                // KEINE Antwort (Netzwerk, Server gestoppt, falsche Adresse)
                throw ApiError(0, "ERRORS.SERVER_UNREACHABLE")
            }
        }

    suspend fun get(url: String) = request("GET", url)

    suspend fun post(url: String, body: JsonElement = JsonObject(emptyMap())) = request("POST", url, body)

    suspend fun put(url: String, body: JsonElement) = request("PUT", url, body)

    suspend fun delete(url: String, body: JsonElement? = null) = request("DELETE", url, body)
}

class ScheduleModel(private val rdvmedecins: RdvMedecins) : ViewModel() {
    // ─── der Zustand: Compose-Zustände, nur hier änderbar ───
    var user by mutableStateOf<User?>(null)
        private set
    var doctors by mutableStateOf(listOf<Person>())
        private set
    var clients by mutableStateOf(listOf<Person>())
        private set
    var doctorId by mutableStateOf<Int?>(null)
        private set
    var day by mutableStateOf(LocalDate.now().plusDays(1).toString()) // morgen, "2026-10-06"
        private set
    var clientId by mutableStateOf<Int?>(null)
        private set
    var schedule by mutableStateOf<Schedule?>(null)
        private set
    var loading by mutableStateOf(false)
        private set
    var message by mutableStateOf<Message?>(null)
        private set
    private val cache = mutableMapOf<String, Schedule>()
    private var reading: Job? = null

    // ─── berechnete Werte ───
    val loggedIn get() = user != null
    val isManager get() = user.let { it != null && it.role != Role.USER }

    // ─── die Aktionen ───
    fun changeUser(u: User?) {
        user = u
        doctors = emptyList()
        clients = emptyList()
        schedule = null
        doctorId = null
        cache.clear() // die Rechte haben sich geändert: Alles wird vergessen
        if (u == null) return
        viewModelScope.launch {
            try {
                doctors = rdvmedecins.getDoctors()
                if (isManager) clients = rdvmedecins.getClients() // ein USER würde 403 erhalten
            } catch (e: ApiError) {
                message = Message(false, e.toString())
            }
        }
    }

    fun chooseDoctor(id: Int?) {
        doctorId = id
        message = null
        readSchedule()
    }

    fun chooseDay(d: LocalDate) {
        day = d.toString()
        message = null
        readSchedule()
    }

    fun chooseClient(id: Int?) {
        clientId = id
    }

    fun book(s: ScheduleSlot) = act("Slot ${s.timeRange} booked.") {
        rdvmedecins.book(day, s.id, if (isManager) clientId else null)
    }

    fun cancel(s: ScheduleSlot) = act("Appointment at ${s.timeRange} cancelled.") { rdvmedecins.cancelAppointment(s.appointmentId!!) }

    // eine Aktion, dann IMMER ein erneutes Lesen des Terminkalenders: Der SERVER ist die Referenz
    private fun act(success: String, action: suspend () -> Unit) {
        viewModelScope.launch {
            message = try {
                action()
                Message(true, success)
            } catch (e: ApiError) {
                Message(false, e.toString())
            }
            readSchedule()
        }
    }

    private fun readSchedule() {
        val id = doctorId ?: return
        val key = "$id/$day"
        // ein älteres Lesen läuft noch: Es wird abgebrochen
        reading?.cancel()
        // zuerst der Cache (oder nichts), dann das Lesen
        schedule = cache[key]
        loading = true
        reading = viewModelScope.launch {
            try {
                val read = rdvmedecins.getSchedule(id, day)
                cache[key] = read
                schedule = read
            } catch (e: ApiError) {
                schedule = null
                message = Message(false, e.toString())
            }
            loading = false
        }
    }
}

@Composable
fun LoginBar(api: Api, onUserChange: (User?) -> Unit) {
    var user by remember { mutableStateOf<User?>(null) }
    var captcha by remember { mutableStateOf<Captcha?>(null) }
    var login by rememberSaveable { mutableStateOf("admin") }
    var password by rememberSaveable { mutableStateOf("") }
    var captchaAnswer by rememberSaveable { mutableStateOf("") }
    var error by remember { mutableStateOf<String?>(null) }
    val scope = rememberCoroutineScope()

    fun changeUser(u: User?) {
        user = u
        onUserChange(u)
    }

    suspend fun newCaptcha() {
        captcha = api.get("/captcha").read<Captcha>()
        captchaAnswer = ""
    }

    // beim Start: sind wir bereits angemeldet?
    LaunchedEffect(Unit) {
        try {
            changeUser(api.get("/auth/me").read<User>())
        } catch (e: ApiError) {
            changeUser(null) // 401: niemand ist angemeldet
            try {
                newCaptcha()
            } catch (e: ApiError) {
                error = "Is the JSON server running (${Config.serverUrl})?"
            }
        }
    }

    suspend fun logIn() {
        error = null
        try {
            val body = buildJsonObject {
                put("login", login.trim())
                put("password", password)
                put("captchaId", captcha?.id)
                put("captcha", captchaAnswer)
            }
            val u = api.post("/auth/login", body).read<User>()
            password = ""
            changeUser(u)
        } catch (e: ApiError) {
            error = e.toString() // "401 LOGIN.ERROR"
            runCatching { newCaptcha() }
        }
    }

    suspend fun logOut() {
        runCatching { api.post("/auth/logout") }
        changeUser(null)
        runCatching { newCaptcha() }
    }

    Card(colors = CardDefaults.cardColors(containerColor = Color(0xFFECEFF1))) {
        val u = user
        if (u != null) {
            Row(Modifier.padding(8.dp), verticalAlignment = Alignment.CenterVertically) {
                Icon(Icons.Default.Person, contentDescription = null)
                Spacer(Modifier.width(8.dp))
                Text("${u.name} (${u.role})", Modifier.weight(1f))
                TextButton(onClick = { scope.launch { logOut() } }) {
                    Icon(Icons.AutoMirrored.Filled.ExitToApp, contentDescription = null)
                    Text("Log out")
                }
            }
        } else {
            Column(Modifier.padding(8.dp)) {
                Row {
                    OutlinedTextField(login, { login = it }, label = { Text("login") }, singleLine = true, modifier = Modifier.weight(1f))
                    Spacer(Modifier.width(8.dp))
                    OutlinedTextField(
                        password,
                        { password = it },
                        label = { Text("password") },
                        singleLine = true,
                        visualTransformation = PasswordVisualTransformation(),
                        modifier = Modifier.weight(1f),
                    )
                }
                Row(verticalAlignment = Alignment.CenterVertically) {
                    // das SVG-Bild des CAPTCHAs, von UNSEREM Server erzeugt (vgl. Beispiel 22)

                    Spacer(Modifier.width(8.dp))

                    IconButton(onClick = { scope.launch { logIn() } }) {
                        Icon(Icons.Default.Check, contentDescription = "Log in")
                    }
                }
                error?.let { Text(it, color = Color.Red) }
            }
        }
    }
}

class Dictionary(context: Context, file: String) {
    // der Inhalt der JSON-Datei, einmal für alle Mal gelesen
    private val root: JsonObject = context.assets.open(file).bufferedReader().use { reader ->
        Json.parseToJsonElement(reader.readText()) as JsonObject
    }

    // "FORM.ADDED" -> root["FORM"]["ADDED"]; null, wenn nicht vorhanden
    private fun find(key: String): String? {
        var node: JsonElement? = root
        for (part in key.split('.')) {
            node = (node as? JsonObject)?.get(part)
        }
        return (node as? JsonPrimitive)?.takeIf { it.isString }?.content
    }

    fun t(key: String, params: Map<String, String> = emptyMap()): String {
        val text = find(key) ?: return key
        // {{name}} -> der Wert des Parameters name (oder {{name}}, wenn er nicht angegeben ist)
        return Regex("\\{\\{(\\w+)\\}\\}").replace(text) { m -> params[m.groupValues[1]] ?: m.value }
    }
}

    return remember(language) { Dictionary(context, "i18n/${prefix}_$language.json") }
}

class ClientsModel(private val api: Api) : ViewModel() {
    var user by mutableStateOf<User?>(null)
        private set
    var clients by mutableStateOf(listOf<Person>())
        private set

    // das Formular: editedId == null -> Erstellung, sonst Aktualisierung
    var editedId by mutableStateOf<Int?>(null)
        private set
    private var version = 0

    // die eingegebenen Werte, nach Feldnamen (die vom Server erwarteten Namen)
    var input by mutableStateOf(empty)
        private set

    // der Fehler jedes Feldes (ein Übersetzungsschlüssel), vom Server zurückgegeben
    var errors by mutableStateOf(mapOf<String, String>())
        private set

    // die allgemeine Meldung des letzten Fehlers und die eines Erfolgs
    var pageError by mutableStateOf<ApiError?>(null)
        private set
    var success by mutableStateOf<KeyMessage?>(null)
        private set

    // Senden an den Server: ein Command (Beispiel 23), das angibt, ob es läuft
    val sending = Command()

    val isAdmin get() = user?.role == Role.ADMIN

    suspend fun onUserChange(u: User?) {
        user = u
        clients = emptyList()
        if (u?.role == Role.ADMIN) reload()
    }

    private suspend fun reload() {
        clients = try {
            api.get("/clients").read()
        } catch (e: ApiError) {
            emptyList()
        }
    }

    private fun clearMessages() {
        errors = emptyMap()
        pageError = null
        success = null
    }

    // ein geändertes Feld: Sein Wert ändert sich, sein Fehler verschwindet
    fun change(field: String, value: String) {
        input = input + (field to value)
        errors = errors - field
    }

    fun newForm() {
        editedId = null
        version = 0
        input = empty
        clearMessages()
    }

    fun edit(c: Person) {
        editedId = c.id
        // die zusammen mit dem Datensatz gelesene VERSION geht mit der Aktualisierung zurück: Hat
        // jemand den Datensatz zwischenzeitlich geändert, antwortet der Server mit 409
        // CONCURRENT_UPDATE
        version = c.version
        input = mapOf("title" to c.title, "firstName" to c.firstName, "lastName" to c.lastName, "login" to (c.login ?: ""), "password" to "")
        clearMessages()
    }

    suspend fun save() {
        clearMessages()
        val body = buildJsonObject {
            for ((field, value) in input) put(field, value)
            put("version", version)
        }
        try {
            val id = editedId
            val client: Person = (if (id == null) api.post("/clients", body) else api.put("/clients/$id", body)).read()
            val key = if (id == null) "FORM.ADDED" else "FORM.UPDATED"
            newForm() // Formular geleert, Meldungen gelöscht ...
            // ... außer dieser
            success = KeyMessage(key, mapOf("name" to "${client.firstName} ${client.lastName}", "login" to (client.login ?: "")))
            reload()
        } catch (e: ApiError) {
            pageError = e // die allgemeine Meldung
            errors = e.fields // der Fehler jedes Feldes
        }
    }

    suspend fun remove(c: Person) {
        clearMessages()
        try {
            api.delete("/clients/${c.id}")
            success = KeyMessage("FORM.DELETED", mapOf("name" to "${c.firstName} ${c.lastName}"))
        } catch (e: ApiError) {
            pageError = e
        }
        reload()
    }

    companion object {
        val empty = mapOf("title" to "Mr", "firstName" to "", "lastName" to "", "login" to "", "password" to "")
    }
}
