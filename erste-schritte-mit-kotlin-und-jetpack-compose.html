@Composable
fun CalculatorPage() {
    // der Zustand: was der Benutzer eingegeben oder gewählt hat
    var label by rememberSaveable { mutableStateOf("Pen") }
    var priceText by rememberSaveable { mutableStateOf("2.5") } // der Text des Feldes, wie eingegeben
    var quantity by rememberSaveable { mutableIntStateOf(1) }
    var vatRate by rememberSaveable { mutableDoubleStateOf(20.0) }
    // der Verlauf der letzten 5 Änderungen der Summe inkl. MwSt.: eine UNVERÄNDERLICHE
    // Liste, die bei jeder Änderung durch eine neue Liste ersetzt wird
    var history by rememberSaveable { mutableStateOf(listOf<String>()) }

    // jeder Ausführung der Funktion neu berechnet werden
    val priceExclVat = price(priceText)
    val totalExclVat = quantity * priceExclVat
    val vatAmount = totalExclVat * vatRate / 100

    // merkt sich die Summe inkl. MwSt. vor der Änderung, wendet die Änderung an und
    // ergänzt den Verlauf, wenn sich die Summe geändert hat
    fun change(update: () -> Unit) {
        val old = inclVat
        update()
        // die Zustandsvariablen haben BEREITS ihren neuen Wert

        if (newTotal != old) {
            history = (listOf("${euros.format(old)} -> ${euros.format(newTotal)}") + history).take(5)
        }
    }

data class Task(val id: Int, val label: String, val done: Boolean = false)

@Composable
fun TasksPage() {
    // von Compose BEOBACHTETE Listen: Das Hinzufügen, Entfernen oder Ersetzen eines
    // Elements löst die Rekomposition aus. (remember und nicht rememberSaveable:
    // Sie gehen verloren, wenn die Activity neu erzeugt wird, vgl. Beispiel 20)
    val tasks = remember {
        mutableStateListOf(Task(1, "Install Android Studio", done = true), Task(2, "Read the chapter on events"))
    }
    val log = remember { mutableStateListOf<String>() }
    var input by remember { mutableStateOf("") }
    var nextId by remember { mutableIntStateOf(3) }
    // der FocusRequester: eine Möglichkeit, dem Feld per Code den Fokus zu geben (den Cursor und die
    // Tastatur des Smartphones)
    val focus = remember { FocusRequester() }

    fun trace(message: String) {
        val time = LocalTime.now().format(DateTimeFormatter.ofPattern("HH:mm"))
        log.add(0, "$time $message")
    }

    // aufgerufen von der Schaltfläche UND von der "OK"-Taste der Tastatur (keyboardActions)
    fun add() {
        val label = input.trim()
        if (label.isEmpty()) {
            return
        }
        tasks.add(Task(nextId++, label))
        trace("added \"$label\"")
        input = ""
        // der Fokus kehrt zum Feld zurück: Die nächste Aufgabe kann eingegeben werden
        focus.requestFocus()
    }

    fun toggle(task: Task, done: Boolean) {
        // wir ERSETZEN die Aufgabe durch eine geänderte Kopie
        tasks[tasks.indexOf(task)] = task.copy(done = done)
        trace("\"${task.label}\": ${if (done) "done" else "to do"}")
    }

    fun remove(task: Task, how: String) {
        tasks.remove(task)
        trace("removed \"${task.label}\" ($how)")
    }

    // die Bildschirmdichte: zur Umrechnung von Pixeln in dp
    val density = LocalDensity.current.density

    ExamplePage("Example 04 - events") {
        Row(verticalAlignment = Alignment.CenterVertically) {
            OutlinedTextField(
                value = input,
                onValueChange = { input = it },
                placeholder = { Text("new task (Escape to clear)") },
                singleLine = true,
                // die "OK"-Taste der Smartphone-Tastatur fügt die Aufgabe hinzu
                keyboardOptions = KeyboardOptions(imeAction = ImeAction.Done),
                keyboardActions = KeyboardActions(onDone = { add() }),
                modifier = Modifier
                    .weight(1f)
                    .focusRequester(focus)
                    // eine physische Tastatur (Emulator, Tablet): Escape leert das Feld
                    .onPreviewKeyEvent { event ->
                        if (event.key == Key.Escape) {
                            input = ""
                            true // das Ereignis ist behandelt
                        } else {
                            false // das Ereignis setzt seinen Weg fort
                        }
                    },
            )
            Spacer(Modifier.width(8.dp))
            Button(onClick = { add() }) { Text("Add") }
        }
        Spacer(Modifier.height(12.dp))
        for (t in tasks) {
            // key(t.id): die IDENTITÄT der Zeile. Der Zustand jeder Zeile (ihr
            // Wischen) folgt IHRER Aufgabe, auch wenn sich die Liste ändert (vgl. den Kurs)
            key(t.id) {
                TaskRow(
                    task = t,
                    onToggle = { done -> toggle(t, done) },
                    onRemove = { how -> remove(t, how) },
                )
            }
        }
        Spacer(Modifier.height(12.dp))
        // pointerInput: Jede Komponente wird empfindlich für Gesten
        Box(
            contentAlignment = Alignment.Center,
            modifier = Modifier
                .fillMaxWidth()
                .height(80.dp)
                .background(Color(0xFFE7F5FF), RoundedCornerShape(8.dp))
                .pointerInput(Unit) {
                    detectTapGestures(
                        // die Position des Drucks relativ zur Komponente (in Pixeln)
                        onPress = { p -> trace("press at (${(p.x / density).roundToInt()}, ${(p.y / density).roundToInt()})") },
                        onDoubleTap = { trace("double tap") },
                        onLongPress = { trace("long press") },
                    )
                },
        ) {
            Text("Touch this area (tap, double tap, long press)")
        }
        Spacer(Modifier.height(16.dp))
        Text("Log", style = MaterialTheme.typography.titleSmall)
        for (line in log) {
            Text(line, fontSize = 12.sp)
        }
    }
}

@Composable
fun TaskRow(task: Task, onToggle: (Boolean) -> Unit, onRemove: (String) -> Unit) {
    // der Wischzustand der Zeile (wo befindet sie sich?)
    val swipe = rememberSwipeToDismissBoxState()
    // SwipeToDismissBox: Das Wischen der Zeile nach links entfernt sie
    SwipeToDismissBox(
        state = swipe,
        enableDismissFromStartToEnd = false,
        // onDismiss: wird aufgerufen, wenn die Zeile ganz weggewischt wurde
        onDismiss = { onRemove("swipe") },
        // was UNTER der Zeile erscheint, wenn sie gewischt wird
        backgroundContent = {
            Box(
                contentAlignment = Alignment.CenterEnd,
                modifier = Modifier.fillMaxSize().background(Color(0xFFFFCDD2)).padding(end = 16.dp),
            ) {
                Icon(Icons.Default.Delete, contentDescription = null)
            }
        },
    ) {
        ListItem(
            // das Kästchen steht LINKS, wie bei einer Aufgabenliste; es
            // gibt seinen neuen Wert (einen Boolean) weiter
            leadingContent = { Checkbox(checked = task.done, onCheckedChange = onToggle) },
            headlineContent = {
                Text(
                    task.label,
                    style = if (task.done) {
                        TextStyle(textDecoration = TextDecoration.LineThrough, color = Color.Gray)
                    } else {
                        TextStyle.Default
                    },
                )
            },
            trailingContent = {
                IconButton(onClick = { onRemove("button") }) {
                    Icon(Icons.Outlined.Delete, contentDescription = "remove", tint = Color.Red)
                }
            },
        )
    }
}

@Composable
fun <T> DropdownList(
    label: String,
    value: T,
    choices: List<T>,
    onChange: (T) -> Unit,
    modifier: Modifier = Modifier,
    text: (T) -> String = { it.toString() },
    // eine Fehlermeldung unter dem Feld (vgl. Beispiele 07 und 25)
    error: String? = null,
) {
    // ist das Menü geöffnet? ein Zustand, der SPEZIFISCH für jede Dropdown-Liste ist
    var open by remember { mutableStateOf(false) }
    ExposedDropdownMenuBox(expanded = open, onExpandedChange = { open = it }, modifier = modifier) {
        // das Feld, das die aktuelle Auswahl anzeigt; menuAnchor macht es zum
        // Ankerpunkt des Menüs (die Berührung öffnet das Menü)
        OutlinedTextField(
            value = text(value),
            onValueChange = {},
            readOnly = true,
            singleLine = true,
            label = { Text(label) },
            trailingIcon = { ExposedDropdownMenuDefaults.TrailingIcon(expanded = open) },
            isError = error != null,
            supportingText = if (error == null) null else { { Text(error) } },
            modifier = Modifier.menuAnchor(ExposedDropdownMenuAnchorType.PrimaryNotEditable).fillMaxWidth(),
        )
        // das Menü: eine Zeile pro Auswahlmöglichkeit
        ExposedDropdownMenu(expanded = open, onDismissRequest = { open = false }) {
            for (c in choices) {
                DropdownMenuItem(
                    text = { Text(text(c)) },
                    onClick = {
                        onChange(c)
                        open = false
                    },
                )
            }
        }
    }
}

//
//     reducer(currentState, action) -> newState

// Der Bildschirm ändert den Zustand nicht mehr: Er BESCHREIBT, was passiert ist, mit einem
// ProfileAction-Objekt (Update(Field.LASTNAME, "Dupont"), Leave(Field.LASTNAME) ...),

enum class Field { TITLE, FIRSTNAME, LASTNAME, AGE, NEWSLETTER, EMAIL, COUNTRY, PRESENTATION }

// die Daten des Formulars: ein UNVERÄNDERLICHES Objekt (alles ist val). Um es
// zu "ändern", erstellen wir eine Kopie davon: profile.copy(lastName = "DUPONT"). Die
// Funktion copy wird von Kotlin für jede "data"-Klasse geschrieben
data class Profile(
    val title: String,
    val firstName: String,
    val lastName: String,
    val age: Int,
    val newsletter: Boolean,
    val email: String,
    val country: String,
    val presentation: String,
) {
    // der Wert eines Feldes, über seinen Namen bezeichnet
    fun value(f: Field): Any = when (f) {
        Field.TITLE -> title
        Field.FIRSTNAME -> firstName
        Field.LASTNAME -> lastName
        Field.AGE -> age
        Field.NEWSLETTER -> newsletter
        Field.EMAIL -> email
        Field.COUNTRY -> country
        Field.PRESENTATION -> presentation
    }

    // eine Kopie, in der EIN Feld, über seinen Namen bezeichnet, einen neuen Wert hat
    fun withValue(f: Field, v: Any): Profile = when (f) {
        Field.TITLE -> copy(title = v as String)
        Field.FIRSTNAME -> copy(firstName = v as String)
        Field.LASTNAME -> copy(lastName = v as String)
        Field.AGE -> copy(age = v as Int)
        Field.NEWSLETTER -> copy(newsletter = v as Boolean)
        Field.EMAIL -> copy(email = v as String)
        Field.COUNTRY -> copy(country = v as String)
        Field.PRESENTATION -> copy(presentation = v as String)
    }

    // das JSON-Objekt des Profils: ein JSON-Feld pro Feld des Formulars
    fun toJson(): JsonObject = JsonObject(
        Field.entries.associate { f ->
            val v = value(f)
            f.name.lowercase() to when (v) {
                is Int -> JsonPrimitive(v) // "is": ein Typtest; v ist dann ein Int
                is Boolean -> JsonPrimitive(v)
                else -> JsonPrimitive(v.toString())
            }
        },
    )
}

// der vollständige Zustand: die Werte und die bereits verlassenen Felder
data class State(val values: Profile, val touched: Set<Field>)

// die möglichen Aktionen. "sealed": Es gibt keine anderen als die
// hier deklarierten, der Compiler weiß das
sealed interface ProfileAction {
    data class Update(val field: Field, val value: Any) : ProfileAction
    data class Leave(val field: Field) : ProfileAction
    data object Uppercase : ProfileAction // ein einzigartiges Objekt: Die Aktion hat keine Daten
    data object Reset : ProfileAction
}

// der Reducer: Er ändert NIE den erhaltenen Zustand, er gibt einen neuen zurück.
// In jedem Zweig des when hat action den Typ des Zweigs ("Smart Cast"):
// action.field ist für ein Update zugänglich
fun reducer(state: State, action: ProfileAction): State = when (action) {
    is ProfileAction.Update -> state.copy(values = state.values.withValue(action.field, action.value))
    is ProfileAction.Leave -> state.copy(touched = state.touched + action.field)
    ProfileAction.Uppercase -> state.copy(
        values = state.values.copy(firstName = state.values.firstName.uppercase(), lastName = state.values.lastName.uppercase()),
    )
    ProfileAction.Reset -> State(initial, emptySet())
}

// das Konto, wie eingegeben
data class Account(
    val title: String,
    val firstName: String,
    val lastName: String,
    val login: String,
    val password: String,
    val confirmation: String,

val titles = listOf("Mr", "Mme", "Melle")
val loginFormat = Regex("^[a-z0-9._-]{1,30}$")
const val PASSWORD_MIN = 6

// die Regeln: eine reine Funktion, die nur von ihrem Argument abhängt. Sie
// gibt für jedes fehlerhafte Feld seine Meldung zurück. buildMap: wir füllen
// eine veränderliche Map, die Funktion gibt eine UNVERÄNDERLICHE Map zurück
fun validate(a: Account): Map<Field, String> = buildMap {
    if (a.title !in titles) put(Field.TITLE, "Choose a title from the list.")
    when {
        a.firstName.isBlank() -> put(Field.FIRSTNAME, "The first name is required.")
        a.firstName.length > 20 -> put(Field.FIRSTNAME, "The first name must not exceed 20 characters.")
    }
    when {
        a.lastName.isBlank() -> put(Field.LASTNAME, "The last name is required.")
        a.lastName.length > 20 -> put(Field.LASTNAME, "The last name must not exceed 20 characters.")
    }
    when {
        a.login.isEmpty() -> put(Field.LOGIN, "The login is required.")
        !loginFormat.matches(a.login) -> put(Field.LOGIN, "Unaccented lower-case letters, digits, dot, hyphen, underscore.")
    }
    when {
        a.password.isEmpty() -> put(Field.PASSWORD, "The password is required.")
        a.password.length < PASSWORD_MIN -> put(Field.PASSWORD, "At least $PASSWORD_MIN characters.")
    }
    // eine Regel, die ZWEI Felder vergleicht
    if (a.confirmation != a.password) put(Field.CONFIRMATION, "The confirmation does not match the password.")
}

                null

import androidx.compose.runtime.getValue

// typealias: ein aussagekräftigerer Name für einen Funktionstyp
typealias Rule = (String) -> String?

// Regel-FABRIKEN: Jede gibt eine Validierungsfunktion zurück
fun required(message: String): Rule = { v -> if (v.isBlank()) message else null }

// eine Regel, die eine andere ERGÄNZT: "?:" (der Elvis-Operator) liefert das
// Ergebnis von other, wenn es nicht null ist, andernfalls das von rest
fun maximum(max: Int, other: Rule): Rule = { v -> other(v) ?: if (v.length > max) "$max characters at most." else null }

class FormField(initialValue: String = "", private val rule: Rule = { null }) {
    // der Zustand des Feldes: von Compose beobachtet, nur HIER änderbar (private set)
    var value by mutableStateOf(initialValue)
        private set
    var error by mutableStateOf<String?>(null)
        private set

    // hat der Benutzer das Feld bereits geändert?
    private var modified = false

    // ein Tastendruck: Das Feld validiert sich selbst, sobald der Benutzer es geändert hat
    fun change(newValue: String) {
        value = newValue
        modified = true
        error = rule(value)
    }

    fun validate(): Boolean {
        error = rule(value)
        return error == null
    }

    // erneut validieren, nur wenn der Benutzer das Feld bereits geändert hat
    fun revalidate() {
        if (modified) validate()
    }
}

// vararg: beliebig viele Felder, Form(firstName, lastName, login ...)
class Form(private vararg val fields: FormField) {
    // map zuerst (ALLE Felder werden validiert und zeigen ihren Fehler an),
    // all danach (sind alle in Ordnung?)
    fun validate(): Boolean = fields.map { it.validate() }.all { it }
}

// ein Zeitfenster, wie von der RdvMedecins-API beschrieben
data class Slot(val startHour: Int, val startMinute: Int, val endHour: Int, val endMinute: Int)

enum class Part { ALL, START, END }

// "8:05" statt "8:5"; private: außerhalb dieser Datei unsichtbar
private fun time(h: Int, m: Int) = "$h:${m.toString().padStart(2, '0')}"

// ein OPTIONALER Parameter mit seinem Standardwert: part = Part.ALL
fun timeRange(s: Slot, part: Part = Part.ALL): String = when (part) {
    Part.START -> time(s.startHour, s.startMinute)
    Part.END -> time(s.endHour, s.endMinute)
    Part.ALL -> "${time(s.startHour, s.startMinute)} - ${time(s.endHour, s.endMinute)}"
}

// ERWEITERUNGEN des Typs String: neue Eigenschaften für ALLE
// Strings. Im Rumpf bezeichnet this den String
// "marie pelissier" -> "Marie Pelissier"; \p{L}: ein Buchstabe, mit oder ohne Akzent
val String.capitalized: String
    get() = Regex("\\p{L}+").replace(this) { m ->
        m.value.lowercase().replaceFirstChar { it.uppercase() }
    }

// "Marie PELISSIER" -> "MP"
val String.initials: String
    get() = split(Regex("\\s+")).filter { it.isNotEmpty() }.joinToString("") { it.first().uppercase() }
    • Zeilen 24–30: ein OPTIONALER Parameter mit seinem Standardwert ([part = Part.ALL]); [padStart(2, '0')]: „8:05“ statt „8:5“;

// Datumsangaben: "vorgefertigte" Formate (SHORT, MEDIUM, LONG, FULL) oder ein Muster
val shortDate: DateTimeFormatter = DateTimeFormatter.ofLocalizedDate(FormatStyle.MEDIUM).withLocale(Locale.US)
val fullDate: DateTimeFormatter = DateTimeFormatter.ofLocalizedDate(FormatStyle.FULL).withLocale(Locale.US)
val timeOnly: DateTimeFormatter = DateTimeFormatter.ofLocalizedTime(FormatStyle.MEDIUM).withLocale(Locale.US)
val frenchDay: DateTimeFormatter = DateTimeFormatter.ofLocalizedDate(FormatStyle.LONG).withLocale(Locale.FRANCE)
val pattern: DateTimeFormatter = DateTimeFormatter.ofPattern("MM/dd/yyyy HH:mm", Locale.US)

// Zahlen und Beträge
val euros: NumberFormat = NumberFormat.getCurrencyInstance(Locale.US).apply { currency = java.util.Currency.getInstance("EUR") }

// apply: konfiguriert das gerade erzeugte Objekt und gibt es zurück
val percent: NumberFormat = NumberFormat.getPercentInstance(Locale.US).apply { minimumFractionDigits = 1 }

@Composable
fun FormattingPage() {
    var now by remember { mutableStateOf(LocalDateTime.now()) }

    // (angezeigter Code, Wert): eine Zeile der Tabelle, als Paar
    val rows = listOf(
        "shortDate.format(now)" to shortDate.format(now),
        "fullDate.format(now)" to fullDate.format(now),
        "timeOnly.format(now)" to timeOnly.format(now),
        "pattern.format(now)" to pattern.format(now),
        "fullDate.format(LocalDate.parse(DAY))" to fullDate.format(LocalDate.parse(DAY)),
        "frenchDay.format(LocalDate.parse(DAY))" to frenchDay.format(LocalDate.parse(DAY)),
        "now.toString()" to now.toString(),
        "euros.format(PRICE)" to euros.format(PRICE),
        "integer.format(PRICE)" to integer.format(PRICE),
        "\"%.1f\".format(Locale.ROOT, PRICE)" to "%.1f".format(Locale.ROOT, PRICE),
        "percent.format(RATE)" to percent.format(RATE),
        "NAME.uppercase()" to NAME.uppercase(),
        "NAME.capitalized" to NAME.capitalized,
        "\"RDV\".lowercase()" to "RDV".lowercase(),
        "languages.take(3).joinToString(\", \")" to languages.take(3).joinToString(", "),
        "JsonObject(...).toString()" to JsonObject(doctor.mapValues { JsonPrimitive(it.value) }).toString(),
        // zwei Transformationen hintereinander: Das Ergebnis der ersten geht an die zweite
        "NAME.capitalized.initials" to NAME.capitalized.initials,
    )

    ExamplePage("Example 09 - formatting") {
        // eine Tabelle: Die erste Spalte nimmt 55 % der Breite ein (weight)
        Row(Modifier.padding(4.dp)) {
            Text("In the code", Modifier.weight(0.55f), fontWeight = FontWeight.Bold)
            Text("Displayed", Modifier.weight(0.45f), fontWeight = FontWeight.Bold)
        }
        // ein Paar kann "destrukturiert" werden: (code, value)
        for ((code, value) in rows) {
            HorizontalDivider()
            Row(Modifier.padding(4.dp)) {
                Text(code, Modifier.weight(0.55f), fontFamily = FontFamily.Monospace, fontSize = 11.sp, color = Color(0xFFC7254E))
                Text(value, Modifier.weight(0.45f))
            }
        }
        Spacer(Modifier.height(16.dp))
        Text("Going through the entries of a Map", style = MaterialTheme.typography.titleSmall)
        for ((key, value) in doctor) {
            Text("$key = $value")
        }
        Spacer(Modifier.height(16.dp))
        Text("A home-made function: timeRange", style = MaterialTheme.typography.titleSmall)
        for (s in slots) {
            Text("${timeRange(s)} (start: ${timeRange(s, Part.START)})")
        }
        Spacer(Modifier.height(16.dp))
        OutlinedButton(onClick = { now = LocalDateTime.now() }) { Text("Refresh the time") }
    }
}
